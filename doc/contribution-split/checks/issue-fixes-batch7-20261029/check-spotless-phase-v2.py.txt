#!/usr/bin/env python3
"""Read an existing fresh invocation; never runs Gradle or changes production source."""
import argparse
import collections
import hashlib
import json
from pathlib import Path
import re

p = argparse.ArgumentParser(description=__doc__)
p.add_argument('--phase-dir', required=True, type=Path)
p.add_argument('--expected-count', type=int, required=True)
p.add_argument('--expected-spotless', required=True,
               help='Exact catalog plugin version for this phase, e.g. 8.10.2 or 8.10.3')
p.add_argument('--baseline-dir', type=Path,
               help='Required for candidate: an independently checked R7 baseline phase')
a = p.parse_args()
out = a.phase_dir.resolve()
invocation = json.loads((out / 'invocation.json').read_text())
completion = json.loads((out / 'completion.json').read_text())
before = json.loads((out / 'target-before.json').read_text())
after = json.loads((out / 'target-after.json').read_text())
events = [json.loads(line) for line in (out / 'events.jsonl').read_text().splitlines()]
log = (out / 'gradle.log').read_text()
nonce, mode = invocation['nonce'], invocation['mode']
source_before, source_after = invocation['source_before'], completion['source_after']
tracked = source_before['tracked_files']
tracked_paths = [x['path'] for x in tracked]
target_paths = [x['path'] for x in before['target_files']]
producer_paths = {
    'build-logic/convention/build/classes/kotlin/main',
    'build-logic/convention/build/pluginDescriptors',
    'build-logic/convention/build/resources/main',
}
expected_producers = {':build-logic:convention:' + x for x in ['compileKotlin', 'pluginDescriptors', 'processResources']}
warning_lines = [line for line in log.splitlines()
                 if 'Producing a file in one build and consuming it in another build' in line
                 and "Task ':spotlessKotlin'" in line]
producers = {m.group(1) for line in warning_lines
             if (m := re.search(r"output of task '([^']+)'", line))}
setvisible_lines = [line for line in log.splitlines()
                    if line.startswith('The Configuration.setVisible(boolean) method has been deprecated.')]
log_lines = log.splitlines()
setvisible_blocks = []
framework_prefixes = ('org.gradle.', 'java.', 'javax.', 'jdk.', 'sun.', 'com.sun.', 'groovy.', 'org.codehaus.groovy.')
for position, line in enumerate(log_lines):
    if not line.startswith('The Configuration.setVisible(boolean) method has been deprecated.'):
        continue
    # Actual --deprecation.trace output is the uninterrupted sequence of
    # tab-prefixed Java stack frames immediately following this message.
    # Never scan an arbitrary later log window for a desired caller.
    stack = []
    for following in log_lines[position + 1:]:
        if not following.startswith('\tat '):
            break
        stack.append(following[len('\tat '):])
    direct_caller = next((frame for frame in stack if not frame.startswith(framework_prefixes)), None)
    component = 'UNKNOWN'
    if direct_caller is not None:
        if direct_caller.startswith('com.android.build.'):
            component = 'AGP'
        elif direct_caller.startswith('com.google.devtools.ksp.'):
            component = 'KSP'
        elif direct_caller.startswith('com.diffplug.gradle.spotless.'):
            component = 'SPOTLESS'
    block_text = '\n'.join([line] + ['\tat ' + frame for frame in stack])
    setvisible_blocks.append({
        'message': line, 'line_number_1based': position + 1,
        'contiguous_stack_sha256': hashlib.sha256(block_text.encode()).hexdigest(),
        'stack_frame_count': len(stack), 'first_nonframework_caller': direct_caller,
        'component': component,
        'gradle_provisioner_frames': [frame for frame in stack
                                       if frame.startswith('com.diffplug.gradle.spotless.GradleProvisioner.')],
    })
setvisible_components = collections.Counter(block['component'] for block in setvisible_blocks)
nonspotless_callers = collections.Counter(block['first_nonframework_caller'] for block in setvisible_blocks
                                         if block['component'] != 'SPOTLESS')
failures = []

def require(condition, message):
    if not condition:
        failures.append(message)

require(nonce == completion['nonce'] == before['nonce'] == after['nonce'], 'Invocation nonce binding mismatch')
require(str(out) == invocation['phase_dir'], 'Invocation belongs to another evidence directory')
require('-Dtrime.spotless.probe.nonce=' + nonce in invocation['argv'], 'Actual argv lacks matching observer nonce')
require('-Dtrime.spotless.probe.out=' + str(out) in invocation['argv'], 'Actual argv lacks matching evidence directory')
require('-Dorg.gradle.deprecation.trace=true' in invocation['argv'], 'Actual argv lacks deprecation stack traces')
require(str(out / 'observer.init.gradle') in invocation['argv'], 'Actual argv did not use frozen observer')
require(hashlib.sha256((out / 'observer.init.gradle').read_bytes()).hexdigest() == invocation['observer_sha256'], 'Frozen observer hash mismatch')
require(hashlib.sha256((out / 'capture-spotless-phase.py').read_bytes()).hexdigest() == invocation['launcher_sha256'], 'Frozen launcher hash mismatch')
require(int((out / 'exit-code.txt').read_text().strip()) == completion['exit_code'] == 0, 'Gradle did not exit zero')
require(source_before == source_after, 'Tracked source, HEAD, status or diff changed during invocation')
require(len(tracked) == a.expected_count, 'Tracked count differs from selected revision')
require(a.expected_count == 293, 'R7 selected upstream revision must still cover all293 tracked Kotlin files')
require(source_before['spotless_dependency']['declared_version'] == a.expected_spotless,
        'Catalog dependency version differs from this phase expectation')
require(len(tracked_paths) == len(set(tracked_paths)), 'Duplicate tracked paths')
require(len(target_paths) == len(set(target_paths)), 'Duplicate runtime target paths')
require(before['target_files'] == tracked, 'Actual formatter target paths/content differs from exact tracked Kotlin inventory')
require(before['target_files'] == after['target_files'], 'Target source content changed during formatter actions')
require(before['target_count'] == after['target_count'] == len(tracked), 'Actual target count differs')
require(before['project_dir'] == after['project_dir'] == invocation['worktree'], 'Observed another project')
require(before['task_path'] == after['task_path'] == ':spotlessKotlin', 'Observed wrong formatter task')
require('SpotlessTaskImpl' in before['task_type'], 'Wrong formatter implementation')
require({x['path'] for x in before['producer_outputs']} == producer_paths and len(before['producer_outputs']) == 3,
        'Producer output set does not equal all three observed producer directories')
require(all(x['file_count'] == len(x['files']) and x['file_count'] > 0 for x in before['producer_outputs']),
        'All three producer output directories must be warm at formatter action time')
require(before['producer_outputs'] == after['producer_outputs'],
        'Producer paths, sizes or hashes changed during formatter actions')
expected_events = [(':spotlessKotlin', 'before'), (':spotlessKotlin', 'after'),
                   (':spotlessKotlinCheck', 'before'), (':spotlessKotlinCheck', 'after')]
require([(x['task_path'], x['event']) for x in events] == expected_events,
        'Missing, duplicate or unordered actual formatter/check task actions')
require([x['sequence'] for x in events] == [1, 2, 3, 4], 'Action sequence mismatch')
require(all(x['nonce'] == nonce and x['project_dir'] == invocation['worktree'] for x in events), 'Event invocation binding mismatch')
require(len({(x['pid'], x['jvm_started_ms']) for x in events}) == 1, 'Actions did not run in same bound JVM')
require(all(x['gradle_version'] == '9.8.1' for x in events), 'Actions did not run on the selected Gradle9.8.1')
require(all(x['monotonic_ns'] < y['monotonic_ns'] for x, y in zip(events, events[1:])), 'Action timestamps not monotonic')
for event in events:
    marker = f"TRIME_SPOTLESS_EVENT nonce={nonce} seq={event['sequence']} task={event['task_path']} event={event['event']}"
    require(log.count(marker) == 1, 'Missing/duplicate actual log marker: ' + marker)
start_marker = f'TRIME_SPOTLESS_EVENT nonce={nonce} seq=1 task=:spotlessKotlin event=before'
end_marker = f'TRIME_SPOTLESS_EVENT nonce={nonce} seq=2 task=:spotlessKotlin event=after'
if log.count(start_marker) == log.count(end_marker) == 1:
    action_log = log.split(start_marker, 1)[1].split(end_marker, 1)[0]
else:
    action_log = ''
require('Not incremental: removing prior outputs' in action_log, 'No bound full/nonincremental formatter action evidence')
require(re.search(r'^> Task :spotlessKotlin\s*$', log, re.M) is not None, 'Formatter task skipped/cache-restored')
require(re.search(r'^> Task :spotlessKotlinCheck\s*$', log, re.M) is not None, 'Check task skipped/cache-restored')
require('BUILD SUCCESSFUL' in log, 'No successful build result')
require(len(setvisible_blocks) == len(setvisible_lines), 'Not all setVisible messages were parsed exactly once')
require(not setvisible_components['UNKNOWN'], 'A setVisible message has missing stack or an unknown direct caller')
require(all(block['stack_frame_count'] >= 3 for block in setvisible_blocks), 'A setVisible direct-caller stack is incomplete')
if mode == 'baseline':
    require(a.baseline_dir is None, 'Baseline must not claim a comparison to another baseline')
    require(a.expected_spotless == '8.10.2', 'R7 baseline must use original Spotless8.10.2')
    require(source_before['head'] == '7fedca06ba5d2ef761a2ae3d44d8a0917db36994',
            'R7 baseline HEAD differs from selected upstream revision')
    require(producers == expected_producers and len(warning_lines) == 3, 'Baseline must reproduce exactly the three known producer warnings')
    require(len(setvisible_lines) == 56, 'Baseline must reproduce all56 actual setVisible messages')
    require(setvisible_components == collections.Counter({'AGP': 52, 'KSP': 3, 'SPOTLESS': 1}),
            'Baseline direct-caller distribution must be AGP52/KSP3/Spotless1 with no unknown source')
    spotless_blocks = [block for block in setvisible_blocks if block['component'] == 'SPOTLESS']
    require(len(spotless_blocks) == 1 and bool(spotless_blocks[0]['gradle_provisioner_frames'])
            and spotless_blocks[0]['first_nonframework_caller'].startswith('com.diffplug.gradle.spotless.GradleProvisioner.'),
            'The single Spotless diagnostic must directly originate in GradleProvisioner')
    baseline_comparison = None
else:
    # #2206 has not landed on this base. Do not require its separate cross-build
    # fix here; record whether a dependency upgrade also changes those warnings.
    require(producers <= expected_producers and len(warning_lines) == len(producers),
            'Candidate has unknown or duplicate root Spotless producer warnings')
    require(len(setvisible_lines) == 55, 'Candidate must retain and account for all55 non-Spotless setVisible messages')
    require(setvisible_components == collections.Counter({'AGP': 52, 'KSP': 3}),
            'Candidate direct-caller distribution must be AGP52/KSP3/Spotless0 with no unknown source')
    require(a.expected_spotless != '8.10.2', 'Candidate must use the selected upgraded Spotless version')
    require(a.baseline_dir is not None, 'Candidate requires --baseline-dir to bind exact A/B Kotlin inputs')
    baseline_comparison = None
    if a.baseline_dir is not None:
        baseline_dir = a.baseline_dir.resolve()
        require(baseline_dir != out, 'Candidate cannot compare itself as baseline')
        baseline_invocation = json.loads((baseline_dir / 'invocation.json').read_text())
        baseline_before = json.loads((baseline_dir / 'target-before.json').read_text())
        baseline_review = json.loads((baseline_dir / 'phase-review-v2.json').read_text())
        require(baseline_invocation['mode'] == baseline_review['mode'] == 'baseline',
                'Comparison evidence is not a baseline phase')
        require(baseline_review['status'] == 'PASS_SCOPED_PHASE', 'Comparison baseline has not passed R7 v2 checker')
        require(baseline_review['nonspotless_direct_caller_multiset'] == dict(sorted(nonspotless_callers.items())),
                'Candidate non-Spotless direct-caller multiset differs from the complete baseline')
        require(baseline_review['checker_sha256'] == hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
                'Baseline was not checked by the same frozen R7 checker')
        require(baseline_review['log_sha256'] == hashlib.sha256((baseline_dir / 'gradle.log').read_bytes()).hexdigest(),
                'Comparison baseline log changed after review')
        require(baseline_invocation['nonce'] == baseline_review['nonce'] == baseline_before['nonce'],
                'Comparison baseline nonce binding mismatch')
        require(baseline_invocation['source_before']['tracked_files'] == tracked,
                'A/B tracked Kotlin paths, sizes or hashes differ')
        require(baseline_before['target_files'] == before['target_files'],
                'A/B actual formatter Kotlin targets differ in paths, sizes or content hashes')
        baseline_comparison = {
            'phase_dir': str(baseline_dir), 'nonce': baseline_invocation['nonce'],
            'log_sha256': baseline_review['log_sha256'],
            'phase_review_sha256': hashlib.sha256((baseline_dir / 'phase-review-v2.json').read_bytes()).hexdigest(),
            'same_all_tracked_kotlin_and_actual_formatter_targets':
                baseline_invocation['source_before']['tracked_files'] == tracked
                and baseline_before['target_files'] == before['target_files'],
            'baseline_setvisible_warning_count': baseline_review['setvisible_warning_count'],
            'nonspotless_direct_caller_multiset_identical':
                baseline_review['nonspotless_direct_caller_multiset'] == dict(sorted(nonspotless_callers.items())),
            'baseline_cross_build_warning_count': baseline_review['warning_line_count'],
            'candidate_cross_build_warning_count': len(warning_lines),
        }
result = {
    'status': 'PASS_SCOPED_PHASE' if not failures else 'FAIL_OR_INCONCLUSIVE',
    'nonce': nonce, 'mode': mode, 'head': source_before['head'],
    'tracked_count': len(tracked), 'target_count': before['target_count'],
    'tracked_counts_by_top_level': dict(collections.Counter(x.split('/')[0] for x in tracked_paths)),
    'tracked_paths_sha256': hashlib.sha256(('\n'.join(tracked_paths) + '\n').encode()).hexdigest(),
    'source_inventory_sha256': hashlib.sha256(json.dumps(tracked, sort_keys=True).encode()).hexdigest(),
    'checker_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    'deprecation_trace_enabled_in_actual_argv': '-Dorg.gradle.deprecation.trace=true' in invocation['argv'],
    'spotless_dependency': source_before['spotless_dependency'],
    'checker_version': 2,
    'setvisible_warning_count': len(setvisible_lines), 'setvisible_diagnostic_blocks': setvisible_blocks,
    'setvisible_components_from_direct_caller': dict(sorted(setvisible_components.items())),
    'spotless_setvisible_warning_count': setvisible_components['SPOTLESS'],
    'nonspotless_direct_caller_multiset': dict(sorted(nonspotless_callers.items())),
    'attribution_rule': 'Only contiguous tab-at stack frames immediately following each message; first non-Gradle/Java framework caller classifies ownership. Unknown/missing stacks fail. All messages remain counted.',
    'baseline_comparison': baseline_comparison,
    'warning_line_count': len(warning_lines), 'producer_tasks': sorted(producers), 'warning_lines': warning_lines,
    'cross_build_warning_scope': 'Separate #2206 fix is not present on this source base; actual0..3 known candidate producer warnings are retained, not asserted eliminated.',
    'producer_output_file_count_actual': sum(x['file_count'] for x in before['producer_outputs']),
    'producer_outputs_unchanged_during_actions': before['producer_outputs'] == after['producer_outputs'],
    'actual_task_action_events': events, 'all_other_warnings_retained_in_original_log': True,
    'log_sha256': hashlib.sha256((out / 'gradle.log').read_bytes()).hexdigest(),
    'failures': failures,
    'scope': 'Bounded local formatter/check phase only; no all-warning, app-build, tests, lint, native, device or CI acceptance claim.',
}
(out / 'phase-review-v2.json').write_text(json.dumps(result, indent=2) + '\n')
print(json.dumps({k: result[k] for k in ['status', 'mode', 'tracked_count', 'target_count', 'setvisible_warning_count', 'spotless_setvisible_warning_count', 'setvisible_components_from_direct_caller', 'warning_line_count', 'producer_output_file_count_actual', 'failures']}))
raise SystemExit(0 if not failures else 1)
